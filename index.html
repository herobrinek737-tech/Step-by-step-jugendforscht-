INTRO

Wie das alles funktioniert

Hier ist die Grundidee: Du trainierst KIs in einem Simulator (kein echter Quad nötig), lädst das fertige Modell dann auf einen Minicomputer (Jetson Nano) auf dem Quad, und der fliegt dann autonom. Du siehst alles live in deinen DJI Googles 3.

KI 1
Freestyle
Lernt Loops, Rolls, Power-Loops und smooth Kurven. Bewertet nach Style-Punkten.
KI 2
Racing
Fliegt Racing-Gates so schnell wie möglich. Optimiert auf minimale Zeit.
KI 3
Target Tracking
Erkennt und verfolgt Menschen, Drohnen, Autos über YOLO-Objekterkennung.
KI 4
Intrusion
Fliegt aggressiv auf ein Ziel zu und präzise hinein. Maximale g-Kräfte.
Für Jugend forscht: Du vergleichst alle 4 KIs in kontrollierten Tests und misst Metriken wie Reaktionszeit, Präzision und Ressourcenverbrauch auf dem Jetson Nano. Das ist dein wissenschaftlicher Beitrag!

PHASE 4

Racing KI trainieren

Wir fangen mit der Racing KI an — sie ist am einfachsten zu messen (Zeit) und macht das Konzept am schnellsten klar.

Die KI lernt durch Reinforcement Learning (RL): Sie probiert zufällig rum, bekommt Punkte für gute Aktionen (Gate getroffen = +50), verliert Punkte für schlechte (Crash = -100). Nach Tausenden von Versuchen wird sie immer besser.

1
Training-Skript erstellen
DATEI: C:\fpv-ki\train_racing.py
import gymnasium as gym
import numpy as np
from stable_baselines3 import PPO
from stable_baselines3.common.env_util import make_vec_env
import airsim

class RacingEnvironment(gym.Env):
    """
    Die KI lebt in dieser Umgebung.
    Sie bekommt:  Sensorwerte der Drohne
    Sie schickt:  Bewegungsbefehle zurück
    """

    def __init__(self):
        super().__init__()
        self.client = airsim.MultirotorClient()
        self.gates = [
            [10,0,-3], [20,5,-4], [30,0,-3],
            [40,-5,-5], [30,-10,-4], [15,-5,-3],
        ]
        self.current_gate = 0
        self.step_count = 0

        # Was die KI steuern kann: [vx, vy, vz, yaw]
        # Jeder Wert zwischen -1 und +1
        self.action_space = gym.spaces.Box(
            low=-1, high=1, shape=(4,), dtype=np.float32
        )

        # Was die KI sehen kann: Position + Geschwindigkeit + nächstes Tor
        self.observation_space = gym.spaces.Box(
            low=-100, high=100, shape=(9,), dtype=np.float32
        )

    def reset(self, seed=None):
        # Neustart: Drohne zurück an Start
        self.client.reset()
        self.client.enableApiControl(True)
        self.client.armDisarm(True)
        self.client.takeoffAsync().join()
        self.current_gate = 0
        self.step_count = 0
        return self._get_obs(), {}

    def step(self, action):
        # KI schickt Befehl → Drohne führt aus
        vx, vy, vz, yaw = action * 10  # Skalieren auf m/s
        self.client.moveByVelocityAsync(vx, vy, vz, 0.1).join()

        obs = self._get_obs()
        reward = self._calc_reward()
        self.step_count += 1

        # Episode beenden wenn: alle Gates geschafft ODER Crash ODER zu viele Steps
        done = (self.current_gate >= len(self.gates)
                or self._crashed()
                or self.step_count > 500)

        return obs, reward, done, False, {}

    def _get_obs(self):
        # Aktuelle Position und Geschwindigkeit auslesen
        state = self.client.getMultirotorState()
        pos = state.kinematics_estimated.position
        vel = state.kinematics_estimated.linear_velocity
        gate = self.gates[min(self.current_gate, len(self.gates)-1)]

        return np.array([
            pos.x_val, pos.y_val, pos.z_val,   # Wo bin ich?
            vel.x_val, vel.y_val, vel.z_val,   # Wie schnell?
            gate[0]-pos.x_val,               # Wie weit zum nächsten Tor?
            gate[1]-pos.y_val,
            gate[2]-pos.z_val,
        ], dtype=np.float32)

    def _calc_reward(self):
        state = self.client.getMultirotorState()
        pos = state.kinematics_estimated.position
        gate = self.gates[min(self.current_gate, len(self.gates)-1)]

        dist = np.sqrt(
            (pos.x_val - gate[0])**2 +
            (pos.y_val - gate[1])**2 +
            (pos.z_val - gate[2])**2
        )

        if dist < 2.0:  # Tor geschafft!
            self.current_gate += 1
            return 50.0

        if self._crashed():
            return -100.0

        # Je näher am Tor, desto besser
        return -dist * 0.1 - 0.5

    def _crashed(self):
        state = self.client.getMultirotorState()
        return state.landed_state == airsim.LandedState.Landed


# ── Training starten ──────────────────────────────
if __name__ == "__main__":
    env = RacingEnvironment()

    # PPO ist ein bewährter RL-Algorithmus (ähnlich wie AlphaGo)
    model = PPO(
        "MlpPolicy",
        env,
        verbose=1,
        learning_rate=3e-4,
        n_steps=2048,
        device="cuda"  # RTX 4060 nutzen!
    )

    print("Training startet... (ca. 2-4 Stunden)")
    model.learn(total_timesteps=500_000)
    model.save("models/racing_ki")
    print("Racing KI gespeichert!")

Target Tracking KI

Die Tracking KI kombiniert zwei Technologien: YOLO (erkennt Objekte im Bild) + Kalman Filter (sagt vorher wohin sich das Ziel bewegt) + PID-Controller (fliegt dorthin).

1

DATEI: test_yolo.py

import cv2

# YOLOv8n = kleinste/schnellste Version

# Webcam-Test (oder Bild-Datei)

    boxes = result.boxes
    for box in boxes:
        cls = result.names[int(box.cls)]

2
Tracking-Controller schreiben
DATEI: tracking_ki.py
from ultralytics import YOLO
import numpy as np
import airsim

model = YOLO("yolov8n.pt")
client = airsim.MultirotorClient()
client.confirmConnection()
client.enableApiControl(True)

FRAME_CENTER = np.array([320, 240])  # 640×480 Kamera
KP = 0.02  # Wie aggressiv der PID reagiert

while True:
    # Bild aus AirSim-Kamera
    response = client.simGetImages([
        airsim.ImageRequest("0", airsim.ImageType.Scene, False, False)
    ])[0]
    img = np.frombuffer(response.image_data_uint8, dtype=np.uint8)
    img = img.reshape(response.height, response.width, 3)

    # YOLO erkennt Objekte
    results = model(img, verbose=False)
    detections = results[0].boxes

    if len(detections) > 0:
        # Bestes Objekt nehmen (höchste Konfidenz)
        best = detections[detections.conf.argmax()]
        x1, y1, x2, y2 = best.xyxy[0]
        center = np.array([(x1+x2)/2, (y1+y2)/2])

        # Fehler = wie weit ist das Ziel vom Bildmittelpunkt?
        error = center - FRAME_CENTER

        # Drohne in Richtung Fehler steuern
        vy = -error[0] * KP  # Links/Rechts
        vz = -error[1] * KP  # Oben/Unten
        vx = 2.0              # Immer langsam vorwärts

        client.moveByVelocityAsync(vx, vy, vz, 0.1)
    else:
        # Kein Ziel → an Ort bleiben
        client.hoverAsync()
PHASE 6

Freestyle & Intrusion KI

Diese beiden KIs bauen auf dem Racing-Code auf. Der Unterschied liegt in der Reward-Funktion — was bekommt die KI als Belohnung?

KI-ModusBelohnung (Reward)StrafeTraining-Zeit
Racing+50 Gate getroffen, +Speed-100 Crash4-6h
Freestyle+G-Kräfte, +Smooth Kurven-Geradeaus fliegen6-8h
Tracking+Ziel im Bild-Zentrum-Ziel verloren3-4h
Intrusion+Nähe zum Ziel, +Geschwindigkeit-Crash zu früh4-5h
Wichtig für Freestyle: Damit die KI "Style" lernt, musst du Manöver in der Reward-Funktion definieren. Ein Power-Loop = Drohne macht eine vertikale 360°-Drehung. Reward = wie nahe kam die Attitude (Lage) dem Ideal-Winkel?

1
OSD via MSP-Protocol

Die DJI Googles nutzen MSP-Protokoll (MultiWii Serial Protocol) für OSD (On-Screen Display). Der Jetson Nano schickt Text und Symbole über dieses Protokoll.

DATEI: googles_overlay.py
import serial
import struct

class GogglesOSD:
    def __init__(self, port="/dev/ttyUSB0", baud=115200):
        self.ser = serial.Serial(port, baud, timeout=1)

    def show_mode(self, mode_name):
        """Aktuellen KI-Modus anzeigen"""
        # MSP_SET_OSD_CONFIG format
        data = mode_name.encode()[:16]  # Max 16 Zeichen
        self._send_msp(84, data)

    def show_target_distance(self, dist_m):
        """Entfernung zum Ziel anzeigen"""
        text = f"TGT {dist_m:.1f}m".encode()
        self._send_msp(84, text)

    def _send_msp(self, cmd, data):
        # MSP Packet bauen
        size = len(data)
        crc = size ^ cmd
        for b in data: crc ^= b
        packet = struct.pack('BBBBs', 36, 77, 60, size, cmd) + data + struct.pack('B', crc)
        self.ser.write(packet)

Waypoints als Linien einzeichnen

Für die Routen-Visualisierung: Der Jetson rendert die geplante Route direkt ins Video-Frame (OpenCV), bevor es an die O4 Air Unit geht.

DATEI: route_overlay.py
import cv2
import numpy as np

def draw_route_on_frame(frame, waypoints_3d, camera_matrix, mode):
    """
    Zeichnet Waypoints als Linien ins FPV-Bild
    waypoints_3d = Liste von (x, y, z) Koordinaten im Weltkoordinatensystem
    """
    frame_copy = frame.copy()

    # Modus-Farben
    colors = {
        "RACING":    (0, 255, 255),  # Gelb
        "FREESTYLE": (255, 0, 255),  # Magenta
        "TRACKING":  (0, 255, 0),    # Grün
        "INTRUSION": (0, 0, 255),    # Rot
    }
    color = colors.get(mode, (255, 255, 255))

    # 3D → 2D projizieren und Linie zeichnen
    prev_pt = None
    for wp in waypoints_3d:
        pt_3d = np.array([wp], dtype=np.float32)
        pt_2d, _ = cv2.projectPoints(pt_3d, np.zeros(3), np.zeros(3), camera_matrix, None)
        pt = (int(pt_2d[0][0][0]), int(pt_2d[0][0][1]))

        cv2.circle(frame_copy, pt, 8, color, 2)
        if prev_pt:
            cv2.line(frame_copy, prev_pt, pt, color, 1)
        prev_pt = pt

    # Modus-Text oben links
    cv2.putText(frame_copy, f"KI: {mode}", (20, 40),
               cv2.FONT_HERSHEY_SIMPLEX, 0.8, color, 2)

    return frame_copy
